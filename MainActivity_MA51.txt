enum class CalendarViewMode { MONTHLY, YEARLY }

@Composable
fun CalendarioApp() {
    val azul = Color(0xFF2196F3)
    MaterialTheme(
        colorScheme = lightColorScheme(
            primary = azul,
            onPrimary = Color.White,
            background = Color.White,
            surface = Color.White,
            onSurface = Color.Black
        )
    ) {
        CalendarioScreen(azul)

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun CalendarioScreen(azul: Color) {
    val context = LocalContext.current
    var currentMonth by remember { mutableStateOf(YearMonth.now()) }
    var currentYear by remember { mutableStateOf(Year.now()) }
    var menuExpanded by remember { mutableStateOf(false) }
    var showCalendarsDialog by remember { mutableStateOf(false) }
    var showAboutDialog by remember { mutableStateOf(false) }
    var viewMode by remember { mutableStateOf(CalendarViewMode.MONTHLY) }
    val today = LocalDate.now()
    var hasCalendarPermission by remember { mutableStateOf(false) }

    var eventsByDate by remember { mutableStateOf(loadEventsFromPrefs(context)) }
    var calendarList by remember { mutableStateOf(listOf<CalendarData>()) }

    // Persistencia de selección de calendarios
    var selectedCalendars by remember { mutableStateOf(loadSelectedCalendars(context)) }

    val requestPermissionLauncher = rememberLauncherForActivityResult(
        contract = ActivityResultContracts.RequestPermission()
    ) { isGranted: Boolean ->
        hasCalendarPermission = isGranted
        if (isGranted) {
            loadCalendars(context) { calendars ->
                calendarList = calendars
                showCalendarsDialog = true
            }
        } else {
            Toast.makeText(context, "Permiso de calendario denegado", Toast.LENGTH_SHORT).show()
        }
    }

    Scaffold(
        topBar = {
            Column(modifier = Modifier.background(azul).statusBarsPadding()) {
                TopAppBar(
                    title = {
                        Text(
                            "Calendario anual",
                            fontSize = 20.sp,
                            color = Color.White,
                            modifier = Modifier.fillMaxWidth(),
                            fontWeight = FontWeight.Bold
                        )
                    },
                    actions = {
                        Box {
                            IconButton(onClick = { menuExpanded = true }) {
                                Icon(Icons.Default.MoreVert, contentDescription = "Menú", tint = Color.White)
                            }
                            DropdownMenu(
                                expanded = menuExpanded,
                                onDismissRequest = { menuExpanded = false },
                                shape = RoundedCornerShape(12.dp),
                                modifier = Modifier.background(Color.White)
                            ) {
                                DropdownMenuItem(
                                    text = { Text("Calendario de festivos", fontSize = 18.sp, modifier = Modifier.padding(8.dp)) },
                                    onClick = {
                                        menuExpanded = false
                                        if (ContextCompat.checkSelfPermission(
                                                context,
                                                android.Manifest.permission.READ_CALENDAR
                                            ) == android.content.pm.PackageManager.PERMISSION_GRANTED
                                        ) {
                                            hasCalendarPermission = true
                                            loadCalendars(context) { calendars ->
                                                calendarList = calendars
                                                showCalendarsDialog = true
                                            }
                                        } else requestPermissionLauncher.launch(android.Manifest.permission.READ_CALENDAR)
                                    }
                                )
                                DropdownMenuItem(
                                    text = { Text("Acerca de", fontSize = 18.sp, modifier = Modifier.padding(8.dp)) },
                                    onClick = {
                                        menuExpanded = false
                                        showAboutDialog = true
                                    }
                                )
                            }
                        }
                    },
                    colors = TopAppBarDefaults.topAppBarColors(
                        containerColor = Color.Transparent,
                        scrolledContainerColor = Color.Transparent,
                        titleContentColor = Color.White,
                        actionIconContentColor = Color.White
                    )
                )
            }
        }
    ) { paddingValues ->
        Column(
            modifier = Modifier.fillMaxSize().padding(paddingValues).padding(12.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                FilledIconButton(
                    onClick = {
                        if (viewMode == CalendarViewMode.MONTHLY) currentMonth = currentMonth.minusMonths(1)
                        else currentYear = currentYear.minusYears(1)
                    },
                    modifier = Modifier.size(44.dp),
                    colors = IconButtonDefaults.filledIconButtonColors(containerColor = Color(0xFFDDDDDD))
                ) { Icon(Icons.Filled.ArrowBack, contentDescription = "Anterior", tint = Color.Black) }

                Button(
                    onClick = {
                        viewMode = if (viewMode == CalendarViewMode.MONTHLY) CalendarViewMode.YEARLY
                        else { currentMonth = YearMonth.now(); CalendarViewMode.MONTHLY }
                    },
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFDDDDDD)),
                    shape = RoundedCornerShape(16.dp),
                    elevation = ButtonDefaults.buttonElevation(defaultElevation = 4.dp)
                ) {
                    Text(
                        if (viewMode == CalendarViewMode.MONTHLY)
                            "${currentMonth.month.getDisplayName(TextStyle.FULL, Locale.getDefault()).replaceFirstChar { it.uppercase() }} ${currentMonth.year}"
                        else "${currentYear.value}",
                        fontSize = 20.sp,
                        color = Color.Black
                    )
                }

                FilledIconButton(
                    onClick = {
                        if (viewMode == CalendarViewMode.MONTHLY) currentMonth = currentMonth.plusMonths(1)
                        else currentYear = currentYear.plusYears(1)
                    },
                    modifier = Modifier.size(44.dp),
                    colors = IconButtonDefaults.filledIconButtonColors(containerColor = Color(0xFFDDDDDD))
                ) { Icon(Icons.Filled.ArrowForward, contentDescription = "Siguiente", tint = Color.Black) }
            }

            Spacer(modifier = Modifier.height(16.dp))

            if (viewMode == CalendarViewMode.MONTHLY) MonthlyCalendar(currentMonth, today, azul, eventsByDate)
            else YearlyCalendar(currentYear, today, azul, eventsByDate) { selectedMonth ->
                currentMonth = selectedMonth
                viewMode = CalendarViewMode.MONTHLY
            }

            if (showCalendarsDialog) {
                FestivosDialog(
                    calendarList = calendarList,
                    selectedCalendars = selectedCalendars,
                    onSelectedChange = {
                        selectedCalendars = it
                        saveSelectedCalendars(context, selectedCalendars)
                    },
                    onApply = {
                        readFestivosFromCalendars(context, selectedCalendars) { festivos ->
                            eventsByDate = festivos
                            saveEventsToPrefs(context, festivos)
                            showCalendarsDialog = false
                        }
                    },
                    onDismiss = { showCalendarsDialog = false }
                )
            }

            if (showAboutDialog) {
                AlertDialog(
                    onDismissRequest = { showAboutDialog = false },
                    title = { Text("Acerca de", fontWeight = FontWeight.Bold, fontSize = 20.sp) },
                    text = {
                        Column {
                            Text("Calendario Anual v1.0", fontSize = 16.sp)
                            Text("Android Studio/ChatGPT", fontSize = 16.sp)
                            Text("Onso/agosto 2025", fontSize = 16.sp)
                            Text("MA51", fontSize = 16.sp)
                        }
                    },
                    confirmButton = {
                        TextButton(onClick = { showAboutDialog = false }) { Text("Cerrar", fontSize = 16.sp) }
                    }
                )
            }
        }
    }
}

// --- Data classes ---
data class CalendarData(val id: Long, val displayName: String)
data class Festivo(val date: LocalDate, val description: String)

// --- Festivos Dialog ---
@Composable
fun FestivosDialog(
    calendarList: List<CalendarData>,
    selectedCalendars: Set<Long>,
    onSelectedChange: (Set<Long>) -> Unit,
    onApply: () -> Unit,
    onDismiss: () -> Unit
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text("Calendario de festivos", fontWeight = FontWeight.Bold, fontSize = 20.sp) },
        text = {
            Column(modifier = Modifier.verticalScroll(rememberScrollState()).heightIn(max = 400.dp)) {
                calendarList.forEach { cal ->
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier.fillMaxWidth().clickable {
                            val newSet = selectedCalendars.toMutableSet()
                            if (newSet.contains(cal.id)) newSet.remove(cal.id)
                            else newSet.add(cal.id)
                            onSelectedChange(newSet)
                        }.padding(vertical = 4.dp)
                    ) {
                        Checkbox(
                            checked = selectedCalendars.contains(cal.id),
                            onCheckedChange = {
                                val newSet = selectedCalendars.toMutableSet()
                                if (it) newSet.add(cal.id) else newSet.remove(cal.id)
                                onSelectedChange(newSet)
                            }
                        )
                        Text(cal.displayName, fontSize = 16.sp)
                    }
                }
            }
        },
        confirmButton = {
            TextButton(onClick = onApply) { Text("Aplicar", fontSize = 16.sp) }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) { Text("Cancelar", fontSize = 16.sp) }
        }
    )
}

// --- Cargar calendarios ---
fun loadCalendars(context: Context, callback: (List<CalendarData>) -> Unit) {
    val resolver = context.contentResolver
    val uri = CalendarContract.Calendars.CONTENT_URI
    val projection = arrayOf(CalendarContract.Calendars._ID, CalendarContract.Calendars.CALENDAR_DISPLAY_NAME)
    val cursor = resolver.query(uri, projection, null, null, null)
    val list = mutableListOf<CalendarData>()
    cursor?.use {
        while (it.moveToNext()) {
            val id = it.getLong(0)
            val name = it.getString(1)
            list.add(CalendarData(id, name))
        }
    }
    callback(list)
}

// --- Leer festivos ---
fun readFestivosFromCalendars(context: Context, selectedCalendars: Set<Long>, callback: (Map<LocalDate, List<String>>) -> Unit) {
    val resolver = context.contentResolver
    val map = mutableMapOf<LocalDate, MutableList<String>>()
    val startMillis = LocalDate.now().minusYears(1).atStartOfDay().atZone(TimeZone.getDefault().toZoneId()).toInstant().toEpochMilli()
    val endMillis = LocalDate.now().plusYears(2).atStartOfDay().atZone(TimeZone.getDefault().toZoneId()).toInstant().toEpochMilli()
    val uri = CalendarContract.Events.CONTENT_URI
    val projection = arrayOf(
        CalendarContract.Events.CALENDAR_ID,
        CalendarContract.Events.DTSTART,
        CalendarContract.Events.TITLE
    )
    val selection = "${CalendarContract.Events.CALENDAR_ID} IN (${selectedCalendars.joinToString(",")})"
    val cursor = resolver.query(uri, projection, selection, null, null)
    cursor?.use {
        while (it.moveToNext()) {
            val calId = it.getLong(0)
            val dtStart = it.getLong(1)
            val title = it.getString(2) ?: ""
            val date = LocalDate.ofEpochDay(dtStart / (24 * 60 * 60 * 1000))
            val list = map.getOrPut(date) { mutableListOf() }
            list.add(title)
        }
    }
    callback(map)
}

// --- Persistencia de eventos/festivos ---
fun saveEventsToPrefs(context: Context, eventsByDate: Map<LocalDate, List<String>>) {
    val prefs = context.getSharedPreferences("events_prefs", Context.MODE_PRIVATE)
    val gson = Gson()
    val mapString = eventsByDate.mapKeys { it.key.toString() }
    prefs.edit().putString("events", gson.toJson(mapString)).apply()
}

fun loadEventsFromPrefs(context: Context): Map<LocalDate, List<String>> {
    val prefs = context.getSharedPreferences("events_prefs", Context.MODE_PRIVATE)
    val gson = Gson()
    val json = prefs.getString("events", null) ?: return emptyMap()
    val type = object : TypeToken<Map<String, List<String>>>() {}.type
    val mapString: Map<String, List<String>> = gson.fromJson(json, type)
    return mapString.mapKeys { LocalDate.parse(it.key) }
}

// --- Persistencia de calendarios seleccionados ---
fun saveSelectedCalendars(context: Context, selectedCalendars: Set<Long>) {
    val prefs = context.getSharedPreferences("events_prefs", Context.MODE_PRIVATE)
    prefs.edit().putStringSet("selectedCalendars", selectedCalendars.map { it.toString() }.toSet()).apply()
}

fun loadSelectedCalendars(context: Context): Set<Long> {
    val prefs = context.getSharedPreferences("events_prefs", Context.MODE_PRIVATE)
    return prefs.getStringSet("selectedCalendars", emptySet())?.map { it.toLong() }?.toSet() ?: emptySet()
}

// --- MONTHLY CALENDAR ---
@Composable
fun MonthlyCalendar(
    currentMonth: YearMonth,
    today: LocalDate,
    azul: Color,
    eventsByDate: Map<LocalDate, List<String>>
) {
    val daysOfWeek = listOf("L","M","X","J","V","S","D")
    var selectedFestivo by remember { mutableStateOf<Pair<LocalDate, String>?>(null) }

    // Encabezado con días de la semana
    Row(
        modifier = Modifier.fillMaxWidth().padding(vertical = 6.dp),
        horizontalArrangement = Arrangement.SpaceAround
    ) {
        daysOfWeek.forEach { day ->
            Text(day, fontSize = 22.sp, fontWeight = FontWeight.Bold, color = Color.Black)
        }
    }

    Spacer(modifier = Modifier.height(8.dp))

    val firstDayOfMonth = currentMonth.atDay(1)
    val firstDayOfWeek = firstDayOfMonth.dayOfWeek.value % 7
    val daysInMonth = currentMonth.lengthOfMonth()
    val totalCells = ((firstDayOfWeek-1 + daysInMonth + 6) / 7) * 7

    // Calendario mensual
    Column {
        for (week in 0 until totalCells step 7) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceAround
            ) {
                for (day in 1..7) {
                    val cellIndex = week + day
                    val dateNumber = cellIndex - (firstDayOfWeek - 1)

                    if (dateNumber in 1..daysInMonth) {
                        val thisDate = currentMonth.atDay(dateNumber)
                        val isToday = thisDate == today
                        val isSunday = thisDate.dayOfWeek == java.time.DayOfWeek.SUNDAY
                        val hasEvent = eventsByDate[thisDate].orEmpty().isNotEmpty()

                        val fontWeight = when {
                            isToday -> FontWeight.Bold
                            hasEvent -> FontWeight.Bold
                            else -> FontWeight.Normal
                        }

                        val textColor = when {
                            isToday -> Color.White
                            hasEvent -> Color.Red
                            isSunday -> Color.Red
                            else -> Color.Black
                        }

                        Box(
                            modifier = Modifier
                                .size(52.dp)
                                .clickable {
                                    if (hasEvent) {
                                        selectedFestivo = thisDate to eventsByDate[thisDate]!!.joinToString("\n")
                                    }
                                },
                            contentAlignment = Alignment.Center
                        ) {
                            if (isToday) {
                                Box(
                                    modifier = Modifier
                                        .size(44.dp)
                                        .background(azul, CircleShape),
                                    contentAlignment = Alignment.Center
                                ) {
                                    Text("$dateNumber", color = Color.White, fontWeight = FontWeight.Bold, fontSize = 20.sp)
                                }
                            } else {
                                Text("$dateNumber", color = textColor, fontWeight = fontWeight, fontSize = 20.sp)
                            }
                        }
                    } else Spacer(modifier = Modifier.size(52.dp))
                }
            }
        }
    }

    Spacer(modifier = Modifier.height(36.dp))

    // Lista de festivos del mes
    val festivosDelMes = eventsByDate.filter { (date, _) ->
        date.year == currentMonth.year && date.month == currentMonth.month
    }.toSortedMap()

    if (festivosDelMes.isNotEmpty()) {
        Column(modifier = Modifier.fillMaxWidth()) {
            Text("Festivos del mes", fontWeight = FontWeight.Bold, fontSize = 18.sp)
            Spacer(modifier = Modifier.height(8.dp))
            festivosDelMes.forEach { (date, events) ->
                events.forEach { event ->
                    Row {
                        Text("${date.dayOfMonth}", color = Color.Red, fontWeight = FontWeight.Bold)
                        Text(": ", color = Color.Black)
                        Text(event)
                    }
                }
            }
        }
    }

    // Diálogo de detalle al pulsar un día con eventos
    selectedFestivo?.let { (date, desc) ->
        AlertDialog(
            onDismissRequest = { selectedFestivo = null },
            title = { Text("${date.dayOfMonth} ${date.month.getDisplayName(TextStyle.FULL, Locale.getDefault())}", fontWeight = FontWeight.Bold) },
            text = { Text(desc, fontSize = 16.sp) },
            confirmButton = {
                TextButton(onClick = { selectedFestivo = null }) { Text("Cerrar", fontSize = 16.sp) }
            }
        )
    }
}

// --- YEARLY CALENDAR ---
@Composable
fun YearlyCalendar(
    currentYear: Year,
    today: LocalDate,
    azul: Color,
    eventsByDate: Map<LocalDate, List<String>>,
    onMonthSelected: (YearMonth) -> Unit
) {
    val months = (1..12).map { YearMonth.of(currentYear.value, it) }
    Column(modifier = Modifier.verticalScroll(rememberScrollState())) {
        for (row in 0 until 6) {
            Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceAround) {
                for (col in 0 until 2) {
                    val index = row * 2 + col
                    if (index < months.size) {
                        Box(
                            modifier = Modifier
                                .weight(1f)
                                .padding(4.dp)
                                .clickable { onMonthSelected(months[index]) },
                            contentAlignment = Alignment.Center
                        ) {
                            MiniMonthCalendar(months[index], today, azul, eventsByDate)
                        }
                    } else Spacer(modifier = Modifier.weight(1f))
                }
            }
        }
    }
}

// --- MINI MONTH CALENDAR ---
@Composable
fun MiniMonthCalendar(
    month: YearMonth,
    today: LocalDate,
    azul: Color,
    eventsByDate: Map<LocalDate, List<String>>
) {
    val daysOfWeek = listOf("L","M","X","J","V","S","D")
    val firstDayOfMonth = month.atDay(1)
    val firstDayOfWeek = firstDayOfMonth.dayOfWeek.value % 7
    val daysInMonth = month.lengthOfMonth()
    val totalCells = ((firstDayOfWeek - 1 + daysInMonth + 6) / 7) * 7

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.spacedBy(2.dp),
        modifier = Modifier
            .fillMaxWidth()
            .background(Color(0xFFF5F5F5), RoundedCornerShape(8.dp))
            .padding(4.dp)
    ) {
        // Mes
        Text(
            month.month.getDisplayName(TextStyle.SHORT, Locale.getDefault()).replaceFirstChar { it.uppercase() },
            fontSize = 14.sp,
            fontWeight = FontWeight.Bold
        )

        // Días de la semana
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceAround
        ) {
            daysOfWeek.forEach { day ->
                Text(day, fontSize = 10.sp, fontWeight = FontWeight.Bold, color = Color.Black)
            }
        }


        // Celdas del mes
        Column(verticalArrangement = Arrangement.spacedBy(2.dp)) {
            for (week in 0 until totalCells step 7) {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceAround
                ) {
                    for (day in 1..7) {
                        val cellIndex = week + day
                        val dateNumber = cellIndex - (firstDayOfWeek - 1)

                        if (dateNumber in 1..daysInMonth) {
                            val thisDate = month.atDay(dateNumber)
                            val isSunday = thisDate.dayOfWeek.value % 7 == 0
                            val hasEvent = eventsByDate[thisDate].orEmpty().isNotEmpty()

                            val textColor = when {
                                isSunday -> Color.Red
                                hasEvent -> Color.Red
                                else -> Color.Black
                            }

                            val fontWeight = when {
                                thisDate == today -> FontWeight.Bold
                                hasEvent -> FontWeight.Bold
                                else -> FontWeight.Normal
                            }

                            Box(
                                modifier = Modifier.size(24.dp),
                                contentAlignment = Alignment.Center
                            ) {
                                if (thisDate == today) {
                                    // Cuadro de fondo azul con offset solo aplicado al fondo
                                    Box(
                                        modifier = Modifier
                                            .size(20.dp)
                                            .offset(y = 3.dp)
                                            .background(azul, RoundedCornerShape(4.dp))
                                    )
                                    // Número centrado sin offset
                                    Text(
                                        "$dateNumber",
                                        fontSize = 12.sp,
                                        color = Color.White,
                                        fontWeight = FontWeight.Bold
                                    )
                                } else {
                                    Text(
                                        "$dateNumber",
                                        fontSize = 12.sp,
                                        color = textColor,
                                        fontWeight = fontWeight
                                    )
                                }
                            }
                        } else Spacer(modifier = Modifier.size(24.dp))
                    }
                }
            }
        }
    }
}
